R · conexión directa a la base
library(DBI)
con <- dbConnect(RPostgres::Postgres(),
                 host = "localhost", dbname = "plataforma",
                 user = "estudiantes",
                 password = Sys.getenv("CLAVE_PLATAFORMA"))
datos <- dbGetQuery(con, "SELECT * FROM {{ tabla }}")
R · descarga por la API
library(httr2)
csv <- request("{{ url_csv }}") |>
  req_headers(`X-API-Key` = Sys.getenv("PLATAFORMA_API_KEY")) |>
  req_perform() |>
  resp_body_string()
datos <- readr::read_csv(I(csv))
Python · pandas
import io, os
import pandas as pd
import requests

r = requests.get("{{ url_csv }}",
                 headers={"X-API-Key": os.environ["PLATAFORMA_API_KEY"]})
r.raise_for_status()
datos = pd.read_csv(io.StringIO(r.content.decode("utf-8-sig")))
